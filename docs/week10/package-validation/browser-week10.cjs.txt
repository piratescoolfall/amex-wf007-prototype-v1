function server(mode,port){
 const log=fs.openSync(p.join(ev,mode+'-server.txt'),'w');
 const args=[p.join(cwd,'node_modules/vite/bin/vite.js'),...(mode==='preview'?['preview']:[]),'--host','127.0.0.1','--port',String(port),'--strictPort'];
 const child=cp.spawn(process.execPath,args,{cwd,env,stdio:['ignore',log,log],windowsHide:true});children.push(child);return child;
}
